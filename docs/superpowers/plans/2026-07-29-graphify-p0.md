```python
def profile_includes(profile, relative_path):
    path = pathlib.PurePosixPath(relative_path)
    if any(path.match(pattern) for pattern in profile["exclude_globs"]):
        return False
    if relative_path in profile["include_files"]:
        return True
    return any(path == root or root in path.parents for root in profile["include_roots"])

def load_profile(config_path, name):
    config = json.loads(config_path.read_text(encoding="utf-8"))
    try:
        return {"name": name, "version": config["version"], **config["profiles"][name]}
    except KeyError as error:
        raise ValueError(f"unknown graph profile: {name}") from error
```

```python
def materialize_profile(repo_root, snapshot, profile):
    counts = {"included": 0, "excluded": 0}
    for source in repo_root.rglob("*"):
        if not source.is_file() or ".git" in source.parts:
            continue
        relative = source.relative_to(repo_root).as_posix()
        if not profile_includes(profile, relative):
            counts["excluded"] += 1
            continue
        target = snapshot / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, target)
        counts["included"] += 1
    return counts
```

```python
def make_manifest(profile, git_sha, file_counts, graph):
    encoded = json.dumps(graph, sort_keys=True, separators=(",", ":")).encode()
    digest = hashlib.sha256(encoded).hexdigest()
    return {
        "build_id": f"{profile}-{git_sha[:12]}-{digest[:12]}",
        "profile": profile, "git_sha": git_sha, "graph_sha256": digest,
        "node_count": len(graph["nodes"]), "edge_count": len(graph["edges"]),
        "file_counts": file_counts,
    }

def validate_build(manifest, graph, report):
    if manifest["node_count"] != len(graph["nodes"]):
        raise ValueError("node count does not match graph")
    if manifest["edge_count"] != len(graph["edges"]):
        raise ValueError("edge count does not match graph")
    if manifest["build_id"] not in report:
        raise ValueError("report does not contain build id")
```

```python
def refresh_profile(repo_root, destination, profile, runner):
    destination.parent.mkdir(parents=True, exist_ok=True)
    staging = pathlib.Path(tempfile.mkdtemp(prefix=f".{profile['name']}-", dir=destination.parent))
    try:
        snapshot = staging / "source"
        counts = materialize_profile(repo_root, snapshot, profile)
        artifacts = staging / "artifacts"
        runner.extract(snapshot, artifacts)
        graph = json.loads((artifacts / "graph.json").read_text(encoding="utf-8"))
        manifest = make_manifest(profile["name"], git_sha(repo_root), counts, graph)
        write_manifest_and_stamps(artifacts, manifest)
        validate_directory(artifacts, manifest, profile)
        previous = destination.parent / f".{destination.name}.previous"
        if destination.exists():
            destination.replace(previous)
        artifacts.replace(destination)
        shutil.rmtree(previous, ignore_errors=True)
    finally:
        shutil.rmtree(staging, ignore_errors=True)
```
